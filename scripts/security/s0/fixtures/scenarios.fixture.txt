// ──────────── v3.1 consumer-edge scenarios (ruleset s0-b05.4) ───────────────
// One factory, one local accessor, four consumers with FOUR DIFFERENT
// behaviours, plus a fifth that is unreadable from its span. The edge model
// must publish all five separately and must not collapse them to one verdict.

function createMixedRepo(owner: Owner | null): MixedRepo {
  const fallback = createSessionMixedRepo(owner);
  async function model(): Promise<MixedModel | null> {
    const db = await getPrisma();
    return db ? ((db as Record<string, unknown>).mixedThing as MixedModel) : null;
  }
  return {
    async list() {
      const m = await model();
      if (!m) return fallback.list();
      try {
        return await m.findMany({ where: { organizationId: owner?.organizationId } });
      } catch {
        return fallback.list();
      }
    },
    async create(input) {
      const m = await model();
      if (!m) return fallback.create(input);
      try {
        return await m.create({ data: input });
      } catch {
        return fallback.create(input);
      }
    },
    async publish(id) {
      const m = await model();
      if (!m) throw new Error("mixed store required");
      return m.update({ where: { id }, data: { status: "published" } });
    },
    async reconcile(id) {
      const m = await model();
      if (!m) return { ok: false, code: "MIXED_STORE_NOT_READY" };
      return { ok: true, row: await m.findUnique({ where: { id } }) };
    },
    async purge(id) {
      return fallback.delete(id);
    },
  };
}

// A SECOND accessor in the SAME FILE, also called `model`, inside a different
// factory. Consumer discovery must not hand this factory's consumers to the one
// above — that double-counting was the v3.0 defect.
function createOtherRepo(): OtherRepo {
  const fallback = createSessionOtherRepo();
  async function model(): Promise<OtherModel | null> {
    const db = await getPrisma();
    return db ? ((db as Record<string, unknown>).otherThing as OtherModel) : null;
  }
  return {
    async count() {
      const m = await model();
      if (!m) return fallback.count();
      return m.count({});
    },
  };
}
